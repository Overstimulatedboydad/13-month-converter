    The 13 months

    Every month has exactly 28 days — four clean weeks. The extra month, Sol, sits between June and July. Your converted month is highlighted.

    How it works

    • Day of year first. Your birthday is counted as a day number (1–365, or 1–366 in a leap year), then dealt into 28-day months: month = which block of 28 it falls in, day = its position in that block.
    • Year Day. Day 365 doesn't belong to any month — it stands alone at the end of the year (Dec 31 on the Gregorian calendar).
    • Leap Day. In a leap year, day 181 (Jun 29) is pulled out as a standalone Leap Day, and every day after it shifts back one so the 28-day months stay aligned.
    • Life Path numbers. Month, day, and year are each reduced to a single digit — except the master numbers 11, 22, and 33, which are kept whole — then added and reduced again. The year is always reduced all the way down (no master numbers kept for the year). The blended number adds the Gregorian and 13-month Life Paths together and reduces once more.
    The original Python code 
    import datetime

MONTHS_13 = [
    "January", "February", "March", "April", "May", "June", "Sol",
    "July", "August", "September", "October", "November", "December"
]

def reduce_num(n, keep_masters=True):
    masters = {11, 22, 33}
    while n > 9 and not (keep_masters and n in masters):
        n = sum(int(d) for d in str(n))
    return n

def convert_birthday(year, month, day):
    bday = datetime.date(year, month, day)
    is_leap = (year % 4 == 0 and (year % 100 != 0 or year % 400 == 0))
    day_of_year = bday.timetuple().tm_yday

    # Gregorian Life Path
    g_lp = reduce_num(
        reduce_num(month) + reduce_num(day) + reduce_num(year, keep_masters=False)
    )

    # Adjust for standalone Leap Day (after June 28 = Day 180 in leap years)
    if is_leap and day_of_year == 181:
        return "Leap Day (Standalone)", g_lp, None, None
    if is_leap and day_of_year > 181:
        day_of_year -= 1
    if day_of_year == 365:
        return "Year Day (Standalone)", g_lp, None, None

    # 13-Month Date & Life Path
    m13_num = ((day_of_year - 1) // 28) + 1
    d13_num = ((day_of_year - 1) % 28) + 1
    m13_name = MONTHS_13[m13_num - 1]

    lp_13 = reduce_num(
        reduce_num(m13_num) + reduce_num(d13_num) + reduce_num(year, keep_masters=False)
    )
    blended_lp = reduce_num(g_lp + lp_13)

    return f"{m13_name} {d13_num}, {year} (Month {m13_num})", g_lp, lp_13, blended_lp

# --- Example Usage ---
date_13, greg_lp, new_lp, combined = convert_birthday(1993, 8, 4)
print(f"13-Month Date: {date_13}")
print(f"Gregorian LP: {greg_lp} | 13-Month LP: {new_lp} | Combined: {combined}")
